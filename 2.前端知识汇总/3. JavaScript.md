### Constructor
```javascript
// Factory Function
function createCircle(radius) {
    return {
        radius,
        draw() {
            console.log('draw');
        }
    };
}
const circle1 = createCircle(1);

// Constructor Function
function Circle(radius) {
    this.radius = radius;
    this.draw = function() {
        console.log('draw');
    }
} 
const circle2 = new Circle(1);
```
constructor 的方式要注意不能去掉new，否则会建立一个global的变量，也就是window级别，整个browser通用的变量。

### Values VS Reference Types
value类型存储的复制的是值
Objects类型复制的是引用
``` javascript
let x= 10;
let y = x;
x = 20;
console.log(x); // 20
console.log(y); // 10
// Primitives are copied by their value
let obj1 = { value: 10 };
let obj2 = obj1;
obj1.value = 20;
console.log(obj1.value); // 20
console.log(obj2.value); // 20
// Objects are copied by their reference

```

### Adding or Removing Properties
```javascript
const circle = new Circle(10);
circle.location = { x: 1 }; // Dynamic nature of objects "radius": 10, "location": {"x": 1}
console.log(circle);
const propertyName = 'center location';
circle[propertyName] = { x: 1 };
console.log(circle); //"radius": 10,"location": {"x": 1},"center location": {"x": 2 }
// Adding and removing properties
delete circle.location;
console.log(circle); //   "radius": 10,"center location": {"x": 2}

```
* 增加属性分为点语法和方括号语法两种，
* 点语法常用，适合属性名固定的场景
* 方括号语法更灵活，支持特殊字符，动态属性名

### Getter and Setter
通过定义精细化的控制，控制属性的读取行为和赋值行为
``` javascript
function Circle(radius) {
    this.radius = radius;
    this.draw = function() {
        console.log('draw');
    }

    Object.defineProperty(this, 'defaultLocation', {
        get: function() {
            return { x: 0, y: 0 };
        },
        set: function(value) {
            if (!value.x || !value.y)
                throw new Error('Invalid location.');
            this._defaultLocation = value;
        }
    });

} 
```


### Practice for above
实现一个计时器，重复计时时可以报错，可以累计的计算时间，可以重置。
```javascript
function stopWatch(){
    let startTime, endTime;
    let duration = 0;
    let running = false;

    this.start= function (){
        if(running){
            throw new Error('Stopwatch has already started.');
        }
        running = true;
        startTime = new Date();
    }
    this.stop= function (){
        if(!running){
            throw new Error('Stopwatch is not started.');
        }
        running = false;
        endTime = new Date();
        const seconds = (endTime - startTime) / 1000;
        duration += seconds;
    }
    Object.defineProperty(this, 'duration', {
        get: function() {
            return duration;
        }
    });
    this.reset= function (){
        startTime = null;
        endTime = null;
        duration = 0;
        running = false;
    }
}
```
